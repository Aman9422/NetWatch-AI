# NetWatch AI — Backend Dependencies
#
# Install with: pip install -r requirements.txt

# Web framework
fastapi>=0.115.0

# ASGI server
uvicorn>=0.30.0

# WebSocket server support for uvicorn (M14). uvicorn is installed *without*
# the ``[standard]`` extra, so it carries no WebSocket implementation of its
# own: without this, an upgrade request to ``/ws/...`` is rejected by the
# server before the application ever sees it.
websockets>=12.0

# Configuration
pydantic-settings>=2.0.0

# Database
sqlalchemy>=2.0.0

# System information
psutil>=5.9.0

# Packet capture (M4)
scapy>=2.6.0

# PDF report rendering (M17). ReportLab is pure Python and pulls no system
# library, so a report can be rendered on any machine the backend itself runs
# on. Its platypus layer is what gives wrapped text, page-breaking tables with
# a repeated header row, and a "Page X of Y" footer — none of which is worth
# hand-writing a PDF writer for.
reportlab>=4.0.0

# Local machine learning for anomaly detection (M22). Isolation Forest is
# scikit-learn's, and it is trained and run on this machine only: no cloud
# inference, no external API and no GPU. The import is lazy (it happens inside
# ``anomaly_detection.model`` at the point a model is built), so a machine
# without this package still captures, detects, alerts and learns baselines --
# only the M22 page reports itself unavailable.
scikit-learn>=1.5.0

# HTTP client for the local LLM adapter (M23). ``app.ai.providers.ollama`` speaks
# Ollama's local HTTP API through ``httpx``: one pooled client for the adapter's
# lifetime, and a **per-call** timeout, so the short health budget (3 s) and the
# long generation budget (60 s) are genuinely different numbers rather than one
# shared setting. That split is what stops a status check from hanging for a
# minute against a stopped runtime.
#
# This was previously a testing-only dependency, because FastAPI's TestClient is
# built on it. M23 promotes it to production because production code now imports
# it; the testing section below keeps listing it as the suites' own requirement.
httpx>=0.27.0

# Testing
pytest>=8.0.0
httpx>=0.27.0
